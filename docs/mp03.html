load_state_population <- function(){
    
    if(!dir.exists(file.path("data", "mp03"))){
        dir.create(file.path("data", "mp03"), showWarnings = FALSE, recursive = TRUE)
    }
    
    # Explicit years (skip 2020)
    years <- c(2015:2019, 2021:2024)
    
    all_data <- list()
    
    for(yr in years){
        
        dest_file <- file.path(
            "data",
            "mp03",
            glue("state_population_{yr}.csv")
        )
        
        if(!file.exists(dest_file)){
            temp <- get_acs(
                variables = c(total_population = "B01003_001"),
                geography = "state",
                year = yr,
                survey = "acs1"
            ) |>
                transmute(
                    State = NAME,
                    Year = yr,
                    `Total Population` = estimate
                )
            
            write_csv(temp, dest_file)
        }
        
        all_data[[as.character(yr)]] <- read_csv(dest_file, show_col_types = FALSE)
    }
    
    bind_rows(all_data)
}

state_to_state_migration_2024 <- function(){
    
    file_name <- file.path(
        "data",
        "mp03",
        "state_to_state_migration_2024.xlsx"
    )
    
    state_lookup <- tibble(Current_Residence = state.name)
    
    read_excel(
        file_name,
        sheet = "Table",
        skip = 6,
        col_names = FALSE
    ) |>
        select(
            Current_Residence = ...1,
            Residence_1_Year_Ago = ...2,
            Estimate = ...3
        ) |>
        fill(Current_Residence) |>
        filter(
            !is.na(Current_Residence),
            !is.na(Residence_1_Year_Ago),
            Current_Residence != "Current Residence",
            Residence_1_Year_Ago != "Residence 1 year ago",
            !str_detect(Residence_1_Year_Ago, "Non-moving"),
            Estimate != "X"
        ) |>
        mutate(
            Estimate = case_when(
                Estimate == "N" ~ "0",
                TRUE ~ as.character(Estimate)
            ),
            Estimate = readr::parse_number(Estimate)
        ) |>
        filter(!is.na(Estimate)) |>
        inner_join(state_lookup, by = "Current_Residence") |>
        inner_join(
            tibble(Residence_1_Year_Ago = state.name),
            by = "Residence_1_Year_Ago"
        ) |>
        group_by(Current_Residence) |>
        summarize(
            `Movers to Current Residence` = sum(Estimate, na.rm = TRUE),
            .groups = "drop"
        ) |>
        arrange(desc(`Movers to Current Residence`))
}

state_to_state_migration_2023 <- function(){
    
    file_name <- file.path(
        "data",
        "mp03",
        "state_to_state_migration_2023.xlsx"
    )
    
    states_with_dc <- c(
        state.name[1:8],
        "District of Columbia",
        state.name[9:50]
    )
    
    stationary <- read_excel(
        file_name,
        sheet = "Table",
        range = "A12:H62",
        col_names = FALSE
    ) |>
        select(
            State = ...1,
            Population_1_Year_And_Over = ...2,
            Same_Residence_1_Year_Ago = ...4,
            Same_State_1_Year_Ago = ...6
        ) |>
        inner_join(
            tibble(State = state.name),
            by = "State"
        ) |>
        mutate(
            Population_1_Year_And_Over = as.character(Population_1_Year_And_Over),
            Same_Residence_1_Year_Ago = as.character(Same_Residence_1_Year_Ago),
            Same_State_1_Year_Ago = as.character(Same_State_1_Year_Ago),
            Population_1_Year_And_Over = case_when(
                Population_1_Year_And_Over == "N" ~ "0",
                Population_1_Year_And_Over == "N/A" ~ "0",
                TRUE ~ Population_1_Year_And_Over
            ),
            Same_Residence_1_Year_Ago = case_when(
                Same_Residence_1_Year_Ago == "N" ~ "0",
                Same_Residence_1_Year_Ago == "N/A" ~ "0",
                TRUE ~ Same_Residence_1_Year_Ago
            ),
            Same_State_1_Year_Ago = case_when(
                Same_State_1_Year_Ago == "N" ~ "0",
                Same_State_1_Year_Ago == "N/A" ~ "0",
                TRUE ~ Same_State_1_Year_Ago
            ),
            Population_1_Year_And_Over = readr::parse_number(Population_1_Year_And_Over),
            Same_Residence_1_Year_Ago = readr::parse_number(Same_Residence_1_Year_Ago),
            Same_State_1_Year_Ago = readr::parse_number(Same_State_1_Year_Ago),
            Population_1_Year_And_Over = as.integer(Population_1_Year_And_Over),
            Same_Residence_1_Year_Ago = as.integer(Same_Residence_1_Year_Ago),
            Same_State_1_Year_Ago = as.integer(Same_State_1_Year_Ago),
            State_1_Year_Ago = State,
            Estimate = Same_State_1_Year_Ago
        ) |>
        select(
            State,
            State_1_Year_Ago,
            Estimate,
            Population_1_Year_And_Over,
            Same_Residence_1_Year_Ago,
            Same_State_1_Year_Ago
        )
    
    flows <- read_excel(
        file_name,
        sheet = "Table",
        range = "J12:DG62",
        col_names = vctrs::vec_interleave(states_with_dc, NA_character_)
    ) |>
        select(any_of(states_with_dc)) |>
        mutate(State = states_with_dc) |>
        pivot_longer(
            cols = -State,
            names_to = "State_1_Year_Ago",
            values_to = "Estimate"
        ) |>
        filter(
            State != "District of Columbia",
            State_1_Year_Ago != "District of Columbia",
            State != State_1_Year_Ago
        ) |>
        inner_join(
            tibble(State = state.name),
            by = "State"
        ) |>
        inner_join(
            tibble(State_1_Year_Ago = state.name),
            by = "State_1_Year_Ago"
        ) |>
        mutate(
            Estimate = as.character(Estimate),
            Estimate = case_when(
                Estimate == "N" ~ "0",
                Estimate == "N/A" ~ "0",
                TRUE ~ Estimate
            ),
            Estimate = readr::parse_number(Estimate),
            Estimate = replace_na(Estimate, 0),
            Estimate = as.integer(Estimate),
            Population_1_Year_And_Over = NA_integer_,
            Same_Residence_1_Year_Ago = NA_integer_,
            Same_State_1_Year_Ago = NA_integer_
        ) |>
        select(
            State,
            State_1_Year_Ago,
            Estimate,
            Population_1_Year_And_Over,
            Same_Residence_1_Year_Ago,
            Same_State_1_Year_Ago
        )
    
    bind_rows(flows, stationary) |>
        arrange(State, State_1_Year_Ago)
}

    if(!dir.exists(file.path("data", "mp03"))){
        dir.create(file.path("data", "mp03"),
                   showWarnings = FALSE,
                   recursive = TRUE)
    }
    
    dest_file <- file.path(
        "data",
        "mp03",
        paste0("state_to_state_migration_", year, ".xlsx")
    )
    
    if(!file.exists(dest_file)){
        request("https://www2.census.gov") |>
            req_url_path(
                "programs-surveys",
                "demo",
                "tables",
                "geographic-mobility",
                as.character(year),
                "state-to-state-migration",
                paste0("State_to_State_Migration_Table_", year, "_T13.xlsx")
            ) |>
            req_perform(dest_file)
    }
    
    dest_file
}

state_to_state_migration_2023 <- function(){
    
    file_name <- download_state_to_state_migration_file(2023)
    
    states_with_dc <- c(
        state.name[1:8],
        "District of Columbia",
        state.name[9:50]
    )
    
    stationary <- read_excel(
        file_name,
        sheet = "Table",
        range = "A12:H62",
        col_names = FALSE
    ) |>
        select(
            State = ...1,
            Population_1_Year_And_Over = ...2,
            Same_Residence_1_Year_Ago = ...4,
            Same_State_1_Year_Ago = ...6
        ) |>
        inner_join(
            tibble(State = state.name),
            by = "State"
        ) |>
        mutate(
            Same_State_1_Year_Ago = as.character(Same_State_1_Year_Ago),
            Same_State_1_Year_Ago = case_when(
                Same_State_1_Year_Ago == "N" ~ "0",
                Same_State_1_Year_Ago == "N/A" ~ "0",
                Same_State_1_Year_Ago == "X" ~ "0",
                TRUE ~ Same_State_1_Year_Ago
            ),
            Same_State_1_Year_Ago = readr::parse_number(Same_State_1_Year_Ago),
            Same_State_1_Year_Ago = replace_na(Same_State_1_Year_Ago, 0),
            Same_State_1_Year_Ago = as.integer(Same_State_1_Year_Ago),
            Residence_1_Year_Ago = State,
            Estimate = Same_State_1_Year_Ago
        ) |>
        select(State, Residence_1_Year_Ago, Estimate)
    
    flows <- read_excel(
        file_name,
        sheet = "Table",
        range = "J12:DG62",
        col_names = vctrs::vec_interleave(states_with_dc, NA_character_)
    ) |>
        select(any_of(states_with_dc)) |>
        mutate(State = states_with_dc) |>
        pivot_longer(
            cols = -State,
            names_to = "Residence_1_Year_Ago",
            values_to = "Estimate"
        ) |>
        filter(
            State != "District of Columbia",
            Residence_1_Year_Ago != "District of Columbia",
            State != Residence_1_Year_Ago
        ) |>
        mutate(
            Estimate = as.character(Estimate),
            Estimate = case_when(
                Estimate == "N" ~ "0",
                Estimate == "N/A" ~ "0",
                Estimate == "X" ~ "0",
                TRUE ~ Estimate
            ),
            Estimate = readr::parse_number(Estimate),
            Estimate = replace_na(Estimate, 0),
            Estimate = as.integer(Estimate)
        ) |>
        select(State, Residence_1_Year_Ago, Estimate)
    
    bind_rows(flows, stationary) |>
        group_by(State, Residence_1_Year_Ago) |>
        summarize(
            Estimate = sum(Estimate, na.rm = TRUE),
            .groups = "drop"
        ) |>
        complete(
            State = state.name,
            Residence_1_Year_Ago = state.name,
            fill = list(Estimate = 0L)
        ) |>
        mutate(Year = 2023L) |>
        select(Year, State, Residence_1_Year_Ago, Estimate) |>
        arrange(State, Residence_1_Year_Ago)
}
