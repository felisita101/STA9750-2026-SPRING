#' Read Olympedia Page With Cache

read_olympedia <- function(url){
    
    if(str_starts(url, "/")){
        url <- str_remove(url, "/")
    }
    
    cache_dir <- fs::path(
        "data",
        "mp04",
        "olympedia_cache"
    )
    
    if(!fs::dir_exists(cache_dir)){
        fs::dir_create(
            cache_dir,
            recurse = TRUE
        )
    }
    
    cache_path <- fs::path(
        cache_dir,
        url |> 
            as.character() |> 
            str_replace_all("/", "_")
    )
    
    if(!fs::file_exists(cache_path)){
        
        src_url <- paste0(
            "https://www.olympedia.org/",
            url
        )
        
        Sys.sleep(5)
        
        request(src_url) |> 
            req_headers(
                `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |> 
            req_perform(path = cache_path)
    }
    
    read_html(cache_path)
}

scrape_edition_disciplines <- function(
    edition_id,
    year,
    city
){
    
    page <- read_olympedia(
        glue("/editions/{edition_id}")
    )
    
    root <- xml2::xml_root(page)
    
    links <- root |> 
        html_elements("a")
    
    tibble(
        discipline_url = links |> 
            html_attr("href"),
        discipline_name = links |> 
            html_text2()
    ) |> 
        filter(
            !is.na(discipline_url),
            str_detect(
                discipline_url,
                glue("^/editions/{edition_id}/sports/")
            )
        ) |> 
        mutate(
            sport_group = case_when(
                str_detect(discipline_name, "Athletics") ~ "Athletics",
                str_detect(discipline_name, "Swimming|Diving|Water Polo|Artistic Swimming|Synchronized Swimming") ~ "Aquatics",
                str_detect(discipline_name, "Gymnastics") ~ "Gymnastics",
                str_detect(discipline_name, "Basketball") ~ "Basketball",
                str_detect(discipline_name, "Cycling") ~ "Cycling",
                str_detect(discipline_name, "Football") ~ "Football",
                str_detect(discipline_name, "Table Tennis") ~ NA_character_,
                str_detect(discipline_name, "Tennis") ~ "Tennis",
                str_detect(discipline_name, "Volleyball") ~ "Volleyball",
                str_detect(discipline_name, "Archery") ~ "Archery",
                str_detect(discipline_name, "Boxing") ~ "Boxing",
                TRUE ~ NA_character_
            )
        ) |> 
        filter(
            !is.na(sport_group),
            sport_group %in% major_summer_sports
        ) |> 
        distinct(
            sport_group,
            discipline_name,
            discipline_url
        ) |> 
        mutate(
            edition_id = edition_id,
            year = year,
            city = city
        )
}

scrape_discipline_medals <- function(
    edition_id,
    year,
    city,
    sport_group,
    discipline_name,
    discipline_url
){
    
    page <- read_olympedia(
        discipline_url
    )
    
    root <- xml2::xml_root(page)
    
    page_text <- root |> 
        html_text2()
    
    edition_roman <- page_text |> 
        str_extract(
            "Number and Year\\s+[IVXLCDM]+"
        ) |> 
        str_remove(
            "Number and Year\\s+"
        )
    
    tables <- root |> 
        html_elements("table")
    
    medal_table <- tables |> 
        map(
            html_table,
            fill = TRUE
        ) |> 
        keep(
            \(x){
                all(
                    c(
                        "NOC",
                        "Gold",
                        "Silver",
                        "Bronze",
                        "Total"
                    ) %in% names(x)
                )
            }
        ) |> 
        first()
    
    if(is.null(medal_table)){
        return(empty_medal_table)
    }
    
    medal_table |> 
        as_tibble(
            .name_repair = "unique_quiet"
        ) |> 
        rename(
            noc = 1
        ) |> 
        transmute(
            competitor_country = str_squish(
                str_remove(
                    noc,
                    "\\s+[A-Z]{3}$"
                )
            ),
            competitor_iso = str_extract(
                noc,
                "[A-Z]{3}$"
            ),
            gold = parse_number(
                as.character(Gold)
            ),
            silver = parse_number(
                as.character(Silver)
            ),
            bronze = parse_number(
                as.character(Bronze)
            ),
            total = parse_number(
                as.character(Total)
            )
        ) |> 
        mutate(
            competitor_country = case_when(
                str_detect(
                    competitor_country,
                    "United States"
                ) ~ "United States",
                str_detect(
                    competitor_country,
                    "Soviet Union"
                ) ~ "Soviet Union",
                str_detect(
                    competitor_country,
                    "People's Republic of China|China"
                ) ~ "China",
                str_detect(
                    competitor_country,
                    "Great Britain"
                ) ~ "Great Britain",
                str_detect(
                    competitor_country,
                    "France"
                ) ~ "France",
                str_detect(
                    competitor_country,
                    "Italy"
                ) ~ "Italy",
                str_detect(
                    competitor_country,
                    "Germany|West Germany|East Germany"
                ) ~ "Germany",
                str_detect(
                    competitor_country,
                    "Japan"
                ) ~ "Japan",
                str_detect(
                    competitor_country,
                    "Hungary"
                ) ~ "Hungary",
                str_detect(
                    competitor_country,
                    "Australia"
                ) ~ "Australia",
                TRUE ~ competitor_country
            )
        ) |> 
        filter(
            competitor_country %in% 
                top_olympic_countries
        ) |> 
        mutate(
            edition_id = edition_id,
            edition_roman = edition_roman,
            edition_url = glue(
                "/editions/{edition_id}"
            ),
            year = year,
            city = city,
            season = "Summer",
            sport_group = sport_group,
            discipline_name = discipline_name,
            discipline_url = discipline_url
        ) |> 
        select(
            edition_id,
            edition_roman,
            edition_url,
            year,
            city,
            season,
            sport_group,
            discipline_name,
            discipline_url,
            competitor_country,
            competitor_iso,
            gold,
            silver,
            bronze,
            total
        )
}

olympic_major_sports_medals <- olympic_editions |> 
    mutate(
        disciplines = pmap(
            list(
                edition_id,
                year,
                city
            ),
            scrape_edition_disciplines
        )
    ) |> 
    select(
        disciplines
    ) |> 
    unnest(
        disciplines
    ) |> 
    mutate(
        medal_data = pmap(
            list(
                edition_id,
                year,
                city,
                sport_group,
                discipline_name,
                discipline_url
            ),
            scrape_discipline_medals
        )
    ) |> 
    pull(
        medal_data
    ) |> 
    bind_rows()

#' All-Sports Olympic Medal Dataset

scrape_edition_medals <- function(
    edition_id,
    year,
    city
){
    
    page <- read_olympedia(
        glue("/editions/{edition_id}")
    )
    
    root <- xml2::xml_root(page)
    
    tables <- root |> 
        html_elements("table")
    
    medal_table <- tables |> 
        map(
            html_table,
            fill = TRUE
        ) |> 
        keep(
            \(x){
                all(
                    c(
                        "NOC",
                        "Gold",
                        "Silver",
                        "Bronze",
                        "Total"
                    ) %in% names(x)
                )
            }
        ) |> 
        first()
    
    if(is.null(medal_table)){
        return(tibble())
    }
    
    medal_table |> 
        as_tibble(
            .name_repair = "unique"
        ) |> 
        rename(
            noc = 1
        ) |> 
        transmute(
            competitor_country = str_squish(
                str_remove(
                    noc,
                    "\\s+[A-Z]{3}$"
                )
            ),
            competitor_iso = str_extract(
                noc,
                "[A-Z]{3}$"
            ),
            gold = parse_number(
                as.character(Gold)
            ),
            silver = parse_number(
                as.character(Silver)
            ),
            bronze = parse_number(
                as.character(Bronze)
            ),
            total = parse_number(
                as.character(Total)
            )
        ) |> 
        mutate(
            competitor_country = case_when(
                str_detect(
                    competitor_country,
                    "United States"
                ) ~ "United States",
                str_detect(
                    competitor_country,
                    "Soviet Union"
                ) ~ "Soviet Union",
                str_detect(
                    competitor_country,
                    "People's Republic of China|China"
                ) ~ "China",
                str_detect(
                    competitor_country,
                    "Great Britain"
                ) ~ "Great Britain",
                str_detect(
                    competitor_country,
                    "France"
                ) ~ "France",
                str_detect(
                    competitor_country,
                    "Italy"
                ) ~ "Italy",
                str_detect(
                    competitor_country,
                    "Germany|West Germany|East Germany"
                ) ~ "Germany",
                str_detect(
                    competitor_country,
                    "Japan"
                ) ~ "Japan",
                str_detect(
                    competitor_country,
                    "Hungary"
                ) ~ "Hungary",
                str_detect(
                    competitor_country,
                    "Australia"
                ) ~ "Australia",
                TRUE ~ competitor_country
            ),
            edition_id = edition_id,
            year = year,
            city = city,
            season = "Summer"
        )
}

    edition_id,
    year,
    city
){
    
    page <- read_olympedia(
        glue("/editions/{edition_id}")
    )
    
    root <- xml2::xml_root(page)
    
    links <- root |> 
        html_elements("a")
    
    tibble(
        discipline_url = links |> 
            html_attr("href"),
        discipline_name = links |> 
            html_text2()
    ) |> 
        filter(
            !is.na(discipline_url),
            str_detect(
                discipline_url,
                glue("^/editions/{edition_id}/sports/")
            ),
            discipline_name != ""
        ) |> 
        distinct(
            discipline_name,
            discipline_url
        ) |> 
        mutate(
            year = year,
            city = city,
            edition_id = edition_id,
            edition_url = glue(
                "/editions/{edition_id}"
            )
        ) |> 
        select(
            year,
            city,
            edition_id,
            edition_url,
            discipline_name,
            discipline_url
        )
}
